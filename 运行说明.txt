# Use vllm. Titan Pascal needs CUDA 11.8, which requires python < 3.13. Set up a conda env.
Use Linux or WSL.
> conda create -n py310 python=3.10
> conda activate py310
> $env:UV_DEFAULT_INDEX="https://pypi.tuna.tsinghua.edu.cn/simple"
> uv pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu118 --system
> uv pip install vllm --system --torch-backend=auto
> $export VLLM_USE_MODELSCOPE=True

Or use docker (require a SM 7.1 GPU)
> docker run --gpus all --env VLLM_USE_MODELSCOPE=True -p 8000:8000 vllm/llm-openai:latest --model Qwen/Qwen3-8B --gpu-memory-utilization 0.8 
# 4B is 2.5G, 8B is 5.2G, 14B is 9.3G
